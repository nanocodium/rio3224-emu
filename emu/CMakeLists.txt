cmake_minimum_required(VERSION 3.16)
project(rio3224_emu CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
file(GLOB_RECURSE SRC CONFIGURE_DEPENDS src/*.cpp)
add_executable(rio-emu ${SRC})
target_include_directories(rio-emu PRIVATE src)
find_package(Threads REQUIRED)
target_link_libraries(rio-emu PRIVATE Threads::Threads)
if(MSVC)
  target_compile_options(rio-emu PRIVATE /W3 /EHsc /utf-8 /wd4244 /wd4267 /wd4996)
  set(CMAKE_CXX_FLAGS_RELEASE "/O2 /DNDEBUG")
  target_compile_definitions(rio-emu PRIVATE _CRT_SECURE_NO_WARNINGS _USE_MATH_DEFINES NOMINMAX WIN32_LEAN_AND_MEAN)
else()
  target_compile_options(rio-emu PRIVATE -Wall -Wno-unused-parameter -Wno-unused-variable)
  set(CMAKE_CXX_FLAGS_RELEASE "-O3 -DNDEBUG")
endif()
if(WIN32)
  target_link_libraries(rio-emu PRIVATE ws2_32 ole32 avrt winmm)
  target_compile_definitions(rio-emu PRIVATE RIO_HAVE_WASAPI=1)
endif()
